$porta = 11434
$wslIp = "172.30.43.212"

New-NetFirewallRule `
  -DisplayName "WSL Port $porta TCP" `
  -Direction Inbound `
  -Action Allow `
  -Protocol TCP `
  -LocalPort $porta

netsh interface portproxy add v4tov4 `
  listenaddress=0.0.0.0 `
  listenport=$porta `
  connectaddress=$wslIp `
  connectport=$porta

  netsh interface portproxy show all
